# M8 多语种输出（翻译）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 菜单栏切换输出语言后，说中文直接注入英/西/德/葡/法/意文字；译砸了退回中文原文并在 HUD 标注。

**Tech Stack:** Swift 6、Swift Testing、MLX Swift（本地）、URLSession（云端）、`./scripts/test.sh`

import Foundation

/// 听写输出的目标语言。
///
/// **本轮只收拉丁字母语种。** 日语、韩语不是技术做不到，是没有判断质量的
/// 手段：闸门「译文里不该有汉字」对日语完全失效（日文本来就用汉字），
/// 而这两种语言用户与实现方都读不了——模型输出一段看起来像日语的东西，
/// 无从判断它是不是把「报价三百」翻成了「报价三千」。而这个工具会把文字
/// **直接注进用户即将发出的窗口**。拉丁字母语种则相反：混语、漏译、
/// 残留汉字肉眼可见。
enum TranslationTarget: String, CaseIterable, Sendable, Equatable {
    /// 不翻译，保持转写原样。
    case chinese = "zh"
    case english = "en"
    case spanish = "es"
    case german = "de"
    case portuguese = "pt"
    case french = "fr"
    case italian = "it"

    /// 选中它是否意味着要走翻译流水线。
    var requiresTranslation: Bool { self != .chinese }

    /// 是否是拉丁字母书写系统。闸门只对这类语言有效，
    /// 将来加日韩时这个标志会变成分流依据。
    var usesLatinScript: Bool { requiresTranslation }

    /// 菜单栏显示用。
    var displayName: String {
        switch self {
        case .chinese: return "中文"
        case .english: return "英文"
        case .spanish: return "西班牙文"
        case .german: return "德文"
        case .portuguese: return "葡萄牙文"
        case .french: return "法文"
        case .italian: return "意大利文"
        }
    }

    /// prompt 里用的语言名。**用英文名而非中文名**：模型对
    /// "Spanish" 的理解比对「西班牙语」稳定，这与 M5 的
    /// `punctuationPrompt` 用具体术语点名是同一个道理。
    var englishName: String {
        switch self {
        case .chinese: return "Chinese"
        case .english: return "English"
        case .spanish: return "Spanish"
        case .german: return "German"
        case .portuguese: return "Portuguese"
        case .french: return "French"
        case .italian: return "Italian"
        }
    }
}
```

    // MARK: - 残留汉字：最硬的一条

    @Test("译文里的汉字被数出来")
    func countsResidualHanCharacters() {
        #expect(TranslationMetrics.residualHanCount("Hello world") == 0)
        #expect(TranslationMetrics.residualHanCount("Hello 世界") == 2)
        // 整句没译＝全是汉字。
        #expect(TranslationMetrics.residualHanCount("你好世界") == 4)
    }

    /// 英文译文里出现「，」和出现汉字是两种不同的失败形态。
    @Test("中文标点残留单独数")
    func countsChinesePunctuation() {
        #expect(TranslationMetrics.residualChinesePunctuationCount("Hello, world.") == 0)
        #expect(TranslationMetrics.residualChinesePunctuationCount("Hello，world。") == 2)
        #expect(TranslationMetrics.residualChinesePunctuationCount("Really？Yes！") == 2)
    }

    // MARK: - 术语保留

    /// 技术术语译过去不该变形。复用 BenchMetrics.technicalTerms 抽词——
    /// 抽而不是手写清单，标准文本改了词表自动跟着改（M5 的做法）。
    @Test("原文里的技术术语在译文里应当原样出现")
    func measuresTermPreservation() {
        let source = "这个服务部署在 Azure 上，用 Terraform 管配置。"
        let good = "This service is deployed on Azure and configured with Terraform."
        let bad = "This service is deployed on Azur and configured with Terra Form."
        #expect(TranslationMetrics.termPreservationRate(source: source, translation: good) == 1.0)
        #expect(TranslationMetrics.termPreservationRate(source: source, translation: bad) < 1.0)
    }

    /// 原文没有术语时返回 1.0 而不是 0——否则口语类会把平均值拖垮，
    /// 而「没有术语可丢」不该被算成「术语全丢了」。
    @Test("原文没有术语时算满分")
    func noTermsMeansPerfect() {
        #expect(TranslationMetrics.termPreservationRate(
            source: "今天天气不错", translation: "Nice weather today") == 1.0)
    }

    // MARK: - 数字保留：报价场景最贵的错

    @Test("阿拉伯数字必须原样保留")
    func measuresDigitPreservation() {
        let source = "起订量五百件，海运要三十五天。"
        // 中文数字译成阿拉伯数字是正常的，所以只看译文里出现了哪些数。
        #expect(TranslationMetrics.digitsIn("MOQ 500 pieces, 35 days by sea") == ["500", "35"])
        #expect(TranslationMetrics.digitsIn("no numbers here").isEmpty)
        _ = source

```swift
import Foundation

/// 翻译质量的机械度量。
///
/// **全部只做机器可判的检查，不碰语义正确性。** 判断「译得对不对」需要
/// 另一个模型，而「用更强的模型检查更强的模型」是无穷回退。这里只回答
/// 「有没有明显没译干净」，与 `CorrectionGate` 同一个定位。
///
/// 素材复用 M5 基准集的 23 条真人录音：标准文本已知，所以下面每一项都能
/// 无人值守地算出来。
enum TranslationMetrics {
    /// CJK 统一表意文字基本区。扩展区在听写场景里见不到，
    /// 漏掉它们只会让这个指标更保守，不会误报。
    private static func isHan(_ c: Character) -> Bool {
        guard c.unicodeScalars.count == 1, let scalar = c.unicodeScalars.first else {
            return false
        }
        return (0x4E00...0x9FFF).contains(scalar.value)
    }

    private static let chinesePunctuation: Set<Character> = [
        "，", "。", "？", "！", "：", "；", "、", "（", "）", "「", "」", "《", "》",
    ]

    /// 译文里残留的汉字个数。**应为 0。** 这是最硬的一条：
    /// 拉丁字母译文里出现汉字，只可能是没译完或者退回了中文。
    static func residualHanCount(_ text: String) -> Int {
        text.filter(isHan).count
    }

    /// 译文里残留的中文标点个数。与残留汉字分开数——
    /// 「英文句子配全角逗号」和「整段没译」是两种不同的失败形态，
    /// 混在一起会看不出是哪种。
    static func residualChinesePunctuationCount(_ text: String) -> Int {
        text.filter { chinesePunctuation.contains($0) }.count
    }

    /// 原文里的技术术语有多少在译文里原样出现。
    ///
    /// 术语用 `BenchMetrics.technicalTerms` 抽而不是手写清单——
    /// 标准文本改了词表自动跟着改，不会漏（M5 的做法）。
    ///
    /// 原文没有术语时返回 1.0：「没有术语可丢」不该被算成「术语全丢了」，
    /// 否则口语类样本会把平均值拖垮。
    static func termPreservationRate(source: String, translation: String) -> Double {
        let terms = BenchMetrics.technicalTerms(in: source)
        guard !terms.isEmpty else { return 1.0 }
        return BenchMetrics.termHitRate(terms: terms, in: translation)
    }

    /// 文本里出现的阿拉伯数字串，按出现顺序。
    static func digitsIn(_ text: String) -> [String] {
        text.split(whereSeparator: { !$0.isNumber })
            .map(String.init)
            .filter { !$0.isEmpty }
    }

    /// 原文里有、译文里没有的阿拉伯数字。
    ///
    /// **数字译错是报价场景最贵的错**，而它恰好是可机器检的少数几件事之一。
    /// 只查原文已经是阿拉伯数字的那些——中文数字（「五百」）译成 `500`
    /// 是正确行为，不该被算成不一致。
    static func lostDigits(source: String, translation: String) -> [String] {
        let inTranslation = Set(digitsIn(translation))
        return digitsIn(source).filter { !inTranslation.contains($0) }
    }

    /// 译文与原文的显著字符数之比。**漏译会明显偏短。**
    ///
    /// 用 `BenchMetrics.significantCharacters` 而不是 `count`：
    /// 标点与空白不该计入，否则中英文的标点密度差异会污染这个比值。
    ///
    /// 原文为空时返回 0 而不是除零崩溃。
    static func lengthRatio(source: String, translation: String) -> Double {
        let sourceCount = BenchMetrics.significantCharacters(source).count
        guard sourceCount > 0 else { return 0 }
        let translationCount = BenchMetrics.significantCharacters(translation).count
        return Double(translationCount) / Double(sourceCount)
    }
}
```

import Testing

/// 门控用**标记文件**而不是环境变量。
///
/// `xcodebuild test` 不把父进程环境变量传给测试运行器——实测会让整个套件
/// 被静默跳过而仍报 TEST SUCCEEDED、退出码 0。M5 期间踩过，`BenchRunnerTests`
/// 与 `CorrectionE2ETests` 都因此改成了标记文件。
@Suite("翻译扫描的门控")
struct TranslationSweepGateTests {

///
/// **默认跳过**——门控是 `Bench/.translate-sweep.json`，由
/// `./scripts/translate-sweep.sh` 写入并在退出时删掉。

    .enabled(if: TranslationSweepConfig.current != nil),
    .serialized

struct TranslationSweepTests {
    @Test("跑全部语言并输出报告")
    func sweep() async throws {
        let config = try #require(TranslationSweepConfig.current)
        let manifest = try BenchManifest.loadFromRepository()
        let translator = MLXTranslator(modelDirectory: config.modelDirectory)
        try await translator.prepare()

        var lines: [String] = [
            "# 翻译对照报告",
            "模型：\(config.modelDirectory.lastPathComponent)",
            "样本：\(manifest.entries.count) 条标准文本",
            "",
        ]

        for target in config.targets {
            var residualHan = 0
            var residualPunct = 0
            var termRates: [Double] = []
            var lostDigitCount = 0
            var ratios: [Double] = []
            var elapsed: [Double] = []

            lines.append("================ \(target.displayName)（\(target.rawValue)）")

            for entry in manifest.entries {
                let started = Date()
                let output = try await translator.translate(entry.text, to: target)
                elapsed.append(Date().timeIntervalSince(started))

                residualHan += TranslationMetrics.residualHanCount(output)
                residualPunct += TranslationMetrics.residualChinesePunctuationCount(output)
                termRates.append(
                    TranslationMetrics.termPreservationRate(
                        source: entry.text, translation: output))
                lostDigitCount += TranslationMetrics.lostDigits(
                    source: entry.text, translation: output).count
                ratios.append(
                    TranslationMetrics.lengthRatio(source: entry.text, translation: output))

                let flag = TranslationMetrics.residualHanCount(output) > 0 ? "❌" : "✅"
                lines.append("\(flag) \(entry.id)")
                lines.append("   原文：\(entry.text)")
                lines.append("   译文：\(output)")
            }

            let n = Double(manifest.entries.count)
            lines.append("""

                【\(target.displayName)】残留汉字=\(residualHan)  \
                残留中文标点=\(residualPunct)  \
                术语保留=\(String(format: "%.2f", termRates.reduce(0, +) / n))  \
                数字丢失=\(lostDigitCount)  \
                长度比=\(String(format: "%.2f", ratios.reduce(0, +) / n))  \
                单条耗时=\(String(format: "%.1f", elapsed.reduce(0, +) / n))秒
                """)
            lines.append("")
        }

        print(report)

        let out = TranslationSweepConfig.markerURL
            .deletingLastPathComponent()
            .appendingPathComponent("translate-sweep.txt")
        try report.write(to: out, atomically: true, encoding: .utf8)

```swift
import Foundation

/// 闸门为什么拦下这次翻译。写进日志，也决定 HUD 说什么。
enum TranslationRejection: Equatable, Sendable {
    /// 译文里有汉字。最硬的一条：拉丁字母译文里出现汉字，
    /// 只可能是没译完或者退回了中文。
    case residualHan(count: Int)
    /// 译文里有中文标点。与残留汉字分开——「英文句子配全角逗号」
    /// 和「整段没译」是两种不同的失败形态。
    case chinesePunctuation(count: Int)
    /// 译文过短，多半漏译。
    case tooShort(ratio: Double)
}

enum TranslationVerdict: Equatable, Sendable {
    case pass
    case reject(TranslationRejection)
}

/// 译文的机械检查。**这一层唯一的机械保障。**
///
/// 与 `CorrectionGate` 同一个定位：只保证「明显错了的挡下来」，
/// 不保证「过了的就是对的」。**故意不做语义正确性检查**——那需要另一个
/// 模型，而「用更强的模型检查更强的模型」是无穷回退。
///
/// 只对拉丁字母目标语言有效。日语用汉字，这套判据整个失效——
/// 这正是本轮不做日韩的原因（见 `TranslationTarget` 的注释）。
struct TranslationGate: Sendable {
    /// 译文/原文的显著字符数之比低于此值即判漏译。
    ///
    /// **这个数必须来自第 0 步的实测**，不是拍的。中译英通常 > 1.0
    /// （英文字符更多），所以下限设在远低于 1 的位置只拦明显的漏译。
    let minimumLengthRatio: Double

    func check(
        source: String, translation: String, target: TranslationTarget
    ) -> TranslationVerdict {
        // **查 usesLatinScript 而不是 requiresTranslation。**
        //
        // 两者今天等价，但语义不同：下面三条判据（残留汉字、中文标点、
        // 长度比）**只对拉丁字母书写系统成立**——日文本来就用汉字，
        // 拿「残留汉字」去判日语译文，正常的译文会被全部拦下。
        //
        // 写成 requiresTranslation 的话，将来加日语时 `TranslationTarget`
        // 那三处 switch 会逼人把 `usesLatinScript` 填成 false（护栏有效），
        // **而闸门仍会把这套判据套上去**——那个护栏就白设了。
        // Task 1 的评审指出了这一点：「这次修复，在它声称保护的地方
        // 买不到任何东西」。
        guard target.usesLatinScript else { return .pass }

        let han = TranslationMetrics.residualHanCount(translation)
        if han > 0 { return .reject(.residualHan(count: han)) }

        let punct = TranslationMetrics.residualChinesePunctuationCount(translation)
        if punct > 0 { return .reject(.chinesePunctuation(count: punct)) }

        let ratio = TranslationMetrics.lengthRatio(source: source, translation: translation)
        if ratio < minimumLengthRatio { return .reject(.tooShort(ratio: ratio)) }

        return .pass
    }
}
```

    ///
    /// **调用方据此决定「这次干脆不翻译」，而不是等模型加载完。**
    /// 加载一次本地模型要几十秒，谁都不该在注入路径上等它。

    ///

import Foundation

/// 这次翻译发生了什么。**只在非正常路径上有值**——
/// 正常译出时是 nil，HUD 什么都不显示。
enum TranslationAnnotation: Equatable, Sendable {
    /// 模型还在加载，本次没翻译。用户可以等一等再说下一句。
    case modelLoading
    /// 推理超时。
    case timedOut
    /// 引擎报错。
    case failed(String)
    /// 选了云端但没配 API key。**单列而不是并进 failed**：
    /// 这一条用户有明确的动作可做（去配 key），其余没有。
    case missingAPIKey
    /// 闸门拦下。
    case rejected(TranslationRejection)

    /// HUD 上显示的一行字。
    ///
    /// **每种都必须有文案**，否则会出现「有标注但界面空白」——
    /// 那正是 M6 ③ 修掉的那类「用户只看到没变化，无从判断是哪种」。
    var hudLabel: String {
        switch self {
        case .modelLoading: return "翻译模型加载中，本次未翻译"
        case .timedOut: return "翻译超时，已输出中文原文"
        case .failed: return "翻译失败，已输出中文原文"
        case .missingAPIKey: return "云端翻译未配置密钥，已输出中文原文"
        case .rejected(.residualHan): return "翻译不完整，已输出中文原文"
        case .rejected(.chinesePunctuation): return "翻译不完整，已输出中文原文"
        case .rejected(.tooShort): return "翻译过短疑似漏译，已输出中文原文"
        }
    }
}

struct TranslationResult: Equatable, Sendable {
    let text: String
    let annotation: TranslationAnnotation?
}

/// 编排翻译：超时、闸门、失败退回。
///
/// **一切失败路径都收敛为「返回中文原文 + 标注」**，本类型不抛错。
/// 这与 M4 的 `CorrectionPipeline` 是同一条原则，也与 M6 确立的
/// 「说过的话不会无声消失」一致：用户宁可自己删掉重说，
/// 也不该发现话没了。
struct TranslationPipeline: Sendable {
    private let translator: any Translator
    private let gate: TranslationGate
    private let timeout: Duration

    init(translator: any Translator, gate: TranslationGate, timeout: Duration) {
        self.translator = translator
        self.gate = gate
        self.timeout = timeout
    }

    /// 后台预热。**只在挂载时调用，不要放在注入路径上**——
    /// 加载一次本地模型要几十秒。
    func prepareTranslator() async throws {
        try await translator.prepare()
    }

    func translate(_ text: String, to target: TranslationTarget) async -> TranslationResult {
        guard target.requiresTranslation else {
            return TranslationResult(text: text, annotation: nil)
        }
        guard !text.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty else {
            return TranslationResult(text: text, annotation: nil)
        }
        // **不在这里等加载。** 加载要几十秒，而调用方带着注入预算，
        // 等下去只会把注入一起拖住（M4 同款处理）。
        guard await translator.isReady else {
            AppLogger.shared.log("翻译跳过：模型尚未就绪")
            return TranslationResult(text: text, annotation: .modelLoading)
        }

        let output: String
        do {
            output = try await withTimeout(timeout) {
                try await translator.translate(text, to: target)
            }
        } catch is TimedOut {
            AppLogger.shared.log("翻译超时：目标=\(target.rawValue)")
            return TranslationResult(text: text, annotation: .timedOut)
        } catch let error as TranslatorError {
            switch error {
            case .missingAPIKey:
                AppLogger.shared.log("翻译失败：云端未配置 API key")
                return TranslationResult(text: text, annotation: .missingAPIKey)
            case .modelNotReady:
                return TranslationResult(text: text, annotation: .modelLoading)
            case .timedOut:
                return TranslationResult(text: text, annotation: .timedOut)
            case .engineFailed(let message):
                AppLogger.shared.log("翻译失败：\(message)")
                return TranslationResult(text: text, annotation: .failed(message))
            }
        } catch {
            AppLogger.shared.log("翻译失败：\(error)")
            return TranslationResult(
                text: text, annotation: .failed(String(describing: error)))
        }

        switch gate.check(source: text, translation: output, target: target) {
        case .pass:
            return TranslationResult(text: output, annotation: nil)
        case .reject(let reason):
            // **记明是哪一条不过**——只说「被拦下」的话，
            // 调参时无从判断是模型的问题还是阈值的问题。
            AppLogger.shared.log("翻译被闸门拦下：\(reason) 目标=\(target.rawValue)")
            return TranslationResult(text: text, annotation: .rejected(reason))
        }
    }
}

/// 超时哨兵。
private struct TimedOut: Error {}

/// 给一段异步工作加超时。
///
/// **用两个非结构化 Task 赛跑，不用 `withThrowingTaskGroup`**——后者会
/// 等所有子任务结束，而推理任务不响应取消，等于超时形同虚设。
/// M4 期间踩过，`CorrectionPipeline.editsWithTimeout` 是同一个写法。
private func withTimeout<T: Sendable>(
    _ duration: Duration,
    _ work: @escaping @Sendable () async throws -> T
) async throws -> T {
    let box = ResultBox<T>()
    let worker = Task { 
        do { await box.finish(.success(try await work())) }
        catch { await box.finish(.failure(error)) }
    }
    let timer = Task {
        try? await Task.sleep(for: duration)
        await box.finish(.failure(TimedOut()))
    }
    defer { timer.cancel() }
    // **不取消 worker**：取消省不下 GPU（推理不响应取消），
    // 却会让「推理已落地」的信号提前发出。M4 的同款结论。
    _ = worker
    return try await box.value().get()
}

private actor ResultBox<T: Sendable> {
    private var stored: Result<T, Error>?
    private var waiters: [CheckedContinuation<Result<T, Error>, Never>] = []

    func finish(_ result: Result<T, Error>) {
        guard stored == nil else { return }
        stored = result
        for waiter in waiters { waiter.resume(returning: result) }
        waiters.removeAll()
    }

    func value() async -> Result<T, Error> {
        if let stored { return stored }
        return await withCheckedContinuation { waiters.append($0) }
    }
}

import Foundation
import MLXLLM
import MLXLMCommon

/// 基于 MLX 的本地翻译器。
///
/// 用 actor 而非 struct：模型加载与推理都不可重入，两次会话同时触发时
/// 第二个必须排队。与 `MLXCorrector` 同一考量。
actor MLXTranslator: Translator {
    /// 模型加载走 `OneShotLoader`：直接判 `container == nil` 有两个坑——
    /// 加载会随调用方的取消一起被掐断，且两次并发调用会各启动一次加载
    /// （判重的 await 之间有空隙）。两条都由闸门解决。
    private let loader: OneShotLoader<ModelContainer>

    /// - Parameter modelDirectory: 本地模型目录。
    ///   **不提供按标识符下载的路径**——`WhisperKitTranscriber` 那个
    ///   `modelName` 默认值会静默联网拉一个无关模型，M5 期间为此修过一轮
    ///   （见 scripts/default-model.sh）。这里从一开始就要求显式给目录。
    init(modelDirectory: URL) {
        loader = OneShotLoader {
            do {
                return try await loadModelContainer(directory: modelDirectory)
            } catch {
                throw TranslatorError.engineFailed(error.localizedDescription)
            }
        }
    }

    var isReady: Bool { get async { await loader.ready != nil } }

    func prepare() async throws { _ = try await loader.value() }

    func translate(_ text: String, to target: TranslationTarget) async throws -> String {
        guard let container = await loader.ready else { throw TranslatorError.modelNotReady }

        // 逐字段赋值而非带标签的构造器：GenerateParameters 的字段在版本间
        // 增删过，逐字段赋值不受参数顺序与标签变化影响（M4 的做法）。
        var parameters = GenerateParameters()
        // 翻译要的是确定性，不是创造力。同一句话每次该得到同样的结果。
        parameters.temperature = 0
        // 译文可能比原文长不少。校正层用 96 是因为替换清单极短，
        // 这里不能照抄——**这个数由第 0 步的实测长度定**。
        parameters.maxTokens = 512

        // **每次新建 ChatSession**：它内部持有对话历史，复用会让上一句
        // 污染下一句。不传 instructions——上游会用赋值覆盖 messages，
        // 系统消息静默丢失（见 TranslationPrompt 的注释）。
        let session = ChatSession(container, generateParameters: parameters)

        // 用流式而非 respond(to:)：后者走同步 generate 循环，全程不检查
        // 取消，超时对它毫无作用。流式版本有 `if Task.isCancelled { break }`。
        var output = ""
        do {
            for try await chunk in session.streamResponse(
                to: TranslationPrompt.user(for: text, target: target)
            ) {
                try Task.checkCancellation()
                output += chunk
            }
        } catch is CancellationError {
            throw TranslatorError.timedOut
        } catch {
            throw TranslatorError.engineFailed(error.localizedDescription)
        }

        return output.trimmingCharacters(in: .whitespacesAndNewlines)
    }
}

```swift
import Foundation

/// 翻译相关的用户偏好。
///
/// 注入 `UserDefaults` 而非直接用 `.standard`：测试需要互不干扰的独立
/// 存储，否则测试之间会互相污染（与 `CorrectionPreference` 同款）。
struct TranslationPreference {
    private enum Key {
        static let outputLanguage = "translation.outputLanguage"
        static let usesCloudBackend = "translation.usesCloudBackend"
        static let modelDirectory = "translation.modelDirectory"
    }

    private let defaults: UserDefaults

    init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
    }

    /// 输出语言。**默认中文，即不翻译**——翻译要额外加载模型并带来延迟，
    /// 用户没主动选之前不该承担这些成本。
    ///
    /// 存稳定字符串而不是枚举序号：加一种语言不该把用户的设置改掉。
    /// 读到不认识的值时退回中文（降级或手改 plist 的情形）。
    var outputLanguage: TranslationTarget {
        get {
            guard let raw = defaults.string(forKey: Key.outputLanguage),
                  let target = TranslationTarget(rawValue: raw)
            else { return .chinese }
            return target
        }
        nonmutating set { defaults.set(newValue.rawValue, forKey: Key.outputLanguage) }
    }

    /// 是否使用云端后端。**默认关闭。**
    ///
    /// 这是用户 2026-08-18 明确的决定：产品原则是尽可能一切留在本地，
    /// 云端不是「要不要做」而是「默认关、用户显式开」。
    var usesCloudBackend: Bool {
        get { defaults.bool(forKey: Key.usesCloudBackend) }
        nonmutating set { defaults.set(newValue, forKey: Key.usesCloudBackend) }
    }

    /// 本地翻译模型目录。
    var modelDirectory: URL? {
        get { defaults.url(forKey: Key.modelDirectory) }
        nonmutating set { defaults.set(newValue, forKey: Key.modelDirectory) }
    }
}
```

        gate: TranslationGate(minimumLengthRatio: 0.3),
        timeout: .seconds(30)

import Foundation
import Testing

@testable import vInput

/// 真模型的端到端体检。门控同 `CorrectionE2ETests`：标记文件
/// `Testing/.translate-e2e-enabled`，内容是 `models/` 下的目录名。
enum TranslationE2EGate {
    static var markerURL: URL {
        URL(fileURLWithPath: #filePath)
            .deletingLastPathComponent()
            .deletingLastPathComponent()
            .deletingLastPathComponent()
            .appendingPathComponent("Testing/.translate-e2e-enabled")
    }

    static var modelDirectory: URL? {
        guard let name = try? String(contentsOf: markerURL, encoding: .utf8)
            .trimmingCharacters(in: .whitespacesAndNewlines), !name.isEmpty
        else { return nil }
        return URL(fileURLWithPath: #filePath)
            .deletingLastPathComponent()
            .deletingLastPathComponent()
            .deletingLastPathComponent()
            .appendingPathComponent("models/\(name)")
    }

    static var isEnabled: Bool {
        guard let dir = modelDirectory else { return false }
        return FileManager.default.fileExists(atPath: dir.path)
    }
}

/// `.serialized`：并发跑 MLX 会让 Metal 崩溃（M4 实测）。
@Suite("翻译端到端", .enabled(if: TranslationE2EGate.isEnabled), .serialized)

    /// 否则每条用例都要重复加载同一份 2 GB 模型。
    private static let translator: MLXTranslator? = {
        guard let dir = TranslationE2EGate.modelDirectory else { return nil }
        return MLXTranslator(modelDirectory: dir)
    }()

    private func makePipeline() async throws -> TranslationPipeline {
        let translator = try #require(Self.translator, "模型不在 models/ 下")
        try await translator.prepare()
        // 超时给够：这里只约束单次推理，加载已在上一行做完。
        return TranslationPipeline(
            translator: translator,
            gate: TranslationGate(minimumLengthRatio: 0.3),
            timeout: .seconds(120)
        )
    }

    @Test("中译英出干净的英文")
    func translatesToEnglish() async throws {
        let pipeline = try await makePipeline()
        let result = await pipeline.translate("今天下午三点开会讨论发版计划。", to: .english)
        #expect(result.annotation == nil, "被拦下了：\(String(describing: result.annotation))")
        #expect(TranslationMetrics.residualHanCount(result.text) == 0)
        print("英文译文：\(result.text)")
    }

```swift
import Foundation
import Security

/// 从 Keychain 读云端翻译的 API key。
///
/// **key 只由用户自己放进 Keychain，不进对话也不进仓库。**
/// 对话记录以明文长期留在 `~/.claude/projects/…/*.jsonl`；本仓库又在
/// OneDrive 下，提交即上云。两条都不是放密钥的地方。
///
/// 用户放 key 的命令（`-w` 后不带值，交互式输入，key 也不进 shell 历史）：
///
/// ```
/// security add-generic-password -a vInput -s vinput-gemini-api-key -w
/// ```
enum TranslationKey {
    static let geminiService = "vinput-gemini-api-key"

    /// 读不到时返回 nil 而不是空字符串——空字符串会让请求带着空 key
    /// 发出去，用户看到的是一个看不懂的 401。
    static func read(service: String) -> String? {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecReturnData as String: true,
            kSecMatchLimit as String: kSecMatchLimitOne,
        ]
        var item: CFTypeRef?
        guard SecItemCopyMatching(query as CFDictionary, &item) == errSecSuccess,
              let data = item as? Data,
              let key = String(data: data, encoding: .utf8),
              !key.isEmpty
        else { return nil }
        return key
    }
}
```

```swift
import Foundation

/// 基于 Gemini 的云端翻译器。**默认不加载**——用户 2026-08-18 的决定：
/// 产品原则是尽可能一切留在本地，云端是「默认关、用户显式开」。
///
/// Gemini 的 API 形状与 Anthropic / OpenAI 的 messages 接口不同
/// （`generateContent`、`contents`/`parts` 结构），所以这里不套那个形状。
struct GeminiTranslator: Translator {
    private let model: String
    private let session: URLSession

    init(model: String = "gemini-2.0-flash", session: URLSession = .shared) {
        self.model = model
        self.session = session
    }

    func prepare() async throws {
        // 云端没有加载成本，但要在这里确认 key 在——否则用户开了云端却
        // 到第一次说话时才发现没配，而那时他正等着文字出现。
        guard TranslationKey.read(service: TranslationKey.geminiService) != nil else {
            throw TranslatorError.missingAPIKey
        }
    }

    func translate(_ text: String, to target: TranslationTarget) async throws -> String {
        guard let key = TranslationKey.read(service: TranslationKey.geminiService) else {
            throw TranslatorError.missingAPIKey
        }

        var request = URLRequest(
            url: URL(string:
                "https://generativelanguage.googleapis.com/v1beta/models/\(model):generateContent"
            )!
        )
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.setValue(key, forHTTPHeaderField: "x-goog-api-key")
        request.httpBody = try JSONSerialization.data(withJSONObject: [
            "contents": [[
                "parts": [["text": TranslationPrompt.user(for: text, target: target)]]
            ]],
            "generationConfig": ["temperature": 0],
        ])

        let (data, response) = try await session.data(for: request)
        guard let http = response as? HTTPURLResponse, http.statusCode == 200 else {
            let code = (response as? HTTPURLResponse)?.statusCode ?? -1
            throw TranslatorError.engineFailed("HTTP \(code)")
        }

        guard let json = try JSONSerialization.jsonObject(with: data) as? [String: Any],
              let candidates = json["candidates"] as? [[String: Any]],
              let content = candidates.first?["content"] as? [String: Any],
              let parts = content["parts"] as? [[String: Any]],
              let output = parts.first?["text"] as? String
        else {
            throw TranslatorError.engineFailed("响应结构不认识")
        }
        return output.trimmingCharacters(in: .whitespacesAndNewlines)
    }
}
```
